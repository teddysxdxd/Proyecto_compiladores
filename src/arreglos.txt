int fibonacci(int n) {
    simon (n <= 1) {
        return n
    }
    return fibonacci(n - 1) + fibonacci(n - 2)
}

int[] nums = [5, 8, 13, 21, 34]
int i = 0
int total = 0
while (i < 5) {
    int r = nums[i] % 2
    simon (r == 0) {
        total = total + nums[i]
    }
    i = i + 1
    simon (total > 50) {
        break
    }
}

print(fibonacci(20))
